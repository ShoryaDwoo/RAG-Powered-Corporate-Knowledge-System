#!/bin/bash
set -e

# ── Logging ───────────────────────────────────────────────────────
exec > >(tee /var/log/user-data.log | logger -t user-data) 2>&1
echo "=== START: $(date) ==="

# ── System update ─────────────────────────────────────────────────
apt-get update -y
apt-get upgrade -y
apt-get install -y curl wget git htop unzip

# ── Install Docker ────────────────────────────────────────────────
curl -fsSL https://get.docker.com | sh
usermod -aG docker ubuntu
systemctl enable docker
systemctl start docker

# ── Install Docker Compose ────────────────────────────────────────
curl -SL https://github.com/docker/compose/releases/latest/download/docker-compose-linux-x86_64 \
  -o /usr/local/bin/docker-compose
chmod +x /usr/local/bin/docker-compose

# ── Install Ollama ────────────────────────────────────────────────
curl -fsSL https://ollama.com/install.sh | sh
systemctl enable ollama
systemctl start ollama

# ── Wait for Ollama to be ready ───────────────────────────────────
echo "Waiting for Ollama to start..."
for i in $(seq 1 30); do
  if curl -s http://localhost:11434/api/tags > /dev/null 2>&1; then
    echo "Ollama is ready after ${i}s"
    break
  fi
  sleep 2
done

# ── Pull compressed Llama 3.2 3B model ───────────────────────────
echo "Pulling llama3.2:3b-instruct-q4_K_M ..."
ollama pull llama3.2:3b-instruct-q4_K_M
echo "Model pull complete"

# ── Verify model loaded ───────────────────────────────────────────
ollama list

# ── Test model ───────────────────────────────────────────────────
echo "Testing model..."
ollama run llama3.2:3b-instruct-q4_K_M "Reply with OK if you are working." --nowordwrap
echo "Model test complete"

# ── Configure Ollama to accept external connections ───────────────
# By default Ollama only listens on localhost
mkdir -p /etc/systemd/system/ollama.service.d
cat <<EOF > /etc/systemd/system/ollama.service.d/override.conf
[Service]
Environment="OLLAMA_HOST=0.0.0.0:11434"
Environment="OLLAMA_MODELS=/usr/share/ollama/.ollama/models"
Environment="OLLAMA_NUM_PARALLEL=2"
Environment="OLLAMA_MAX_LOADED_MODELS=1"
EOF

systemctl daemon-reload
systemctl restart ollama

# ── Wait for restart ──────────────────────────────────────────────
sleep 5
echo "Ollama restarted and listening on 0.0.0.0:11434"

# ── Install Python 3.11 + pip ─────────────────────────────────────
apt-get install -y python3.11 python3.11-venv python3-pip
python3.11 -m pip install --upgrade pip

# ── Setup app directory ───────────────────────────────────────────
mkdir -p /app
chown ubuntu:ubuntu /app

# ── Write a quick health-check script ────────────────────────────
cat <<'EOF' > /usr/local/bin/check-ollama
#!/bin/bash
echo "=== Ollama Status ==="
systemctl status ollama --no-pager
echo ""
echo "=== Loaded Models ==="
ollama list
echo ""
echo "=== API Health ==="
curl -s http://localhost:11434/api/tags | python3 -m json.tool
EOF
chmod +x /usr/local/bin/check-ollama

echo "=== DONE: $(date) ==="
echo "Ollama is running with llama3.2:3b-instruct-q4_K_M"
echo "API available at http://<EC2-PUBLIC-IP>:11434"
echo "Run 'check-ollama' to verify status"
